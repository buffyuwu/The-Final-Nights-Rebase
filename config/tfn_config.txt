## Hostname or IP that clients use to reach the Babylon WebSocket server.
## Set to 127.0.0.1 for local testing. Change to your public IP or domain for hosted servers.
babylon_server_host 127.0.0.1

## uncomment to hide ckeys from mentors
#MENTORS_MOBNAME_ONLY

## uncomment to use mentors.txt instead of SQL
#MENTOR_LEGACY_SYSTEM
